# Anestheo — https://anestheo.com/
#
# The application is served from the document root. Everything behind a login
# is listed below not because robots.txt protects it — it does not, and must
# never be relied on for that — but because there is nothing there for a
# crawler to index. Access is enforced by Supabase RLS and the page auth
# guards; this file only avoids wasting crawl budget on pages that render a
# redirect to the homepage for anyone who is not signed in.
#
# Patient-identifying URLs (the questionnaire token link) are excluded for a
# stronger reason: a token in a URL must never end up in a search index.

User-agent: *

Disallow: /q.html

# ── Videos: the preserved V1 page ───────────────────────────────────────────
# /videos.html is served, internally, from videos-v1/. Both addresses return
# the same bytes, so the folder is excluded to keep one canonical URL. Only
# /videos.html should ever be indexed. Remove this when the port task lands and
# the folder goes away.
Disallow: /videos-v1/
Disallow: /videos-v1.html

# ── Legacy URLs — keep these CRAWLABLE ──────────────────────────────────────
# Every /v2/ URL, and every retired V1 page, 301s to its replacement. The
# redirect is what consolidates ranking signal onto the new URL, so the old
# paths must stay crawlable — a Disallow would stop crawlers ever seeing the
# 301, and the old URLs would linger in the index instead of being replaced.
#
# Do NOT add "Disallow: /v2/", and do not disallow auth.html, register.html,
# doctor.html, patient.html, calculator.html, calculators.html, clinical.html
# or legal.html. They are redirects now, and a redirect has to be readable to
# do its job.
